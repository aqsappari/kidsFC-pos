"use server";

import { db } from "@/db";
import { orders, ingredients, products } from "@/db/schema";
import { count, eq, sql } from "drizzle-orm";

export async function getAdminDashboardMetrics() {
  // 1. Fetch all completed orders
  const allOrders = await db.query.orders.findMany({
    where: eq(orders.status, "completed"),
  });

  // 2. Fetch all ingredients to check for low stock
  const allIngredients = await db.query.ingredients.findMany();

  // 3. Fetch all products count
  const allProducts = await db.query.products.findMany();

  // Calculate today's sales (compare ISO date string prefix YYYY-MM-DD)
  const todayStr = new Date().toISOString().split("T")[0];
  let todaySales = 0;
  let todayOrdersCount = 0;

  for (const ord of allOrders) {
    if (ord.createdAt.startsWith(todayStr)) {
      todaySales += ord.totalAmount;
      todayOrdersCount += 1;
    }
  }

  // Count low-stock ingredients
  const lowStockIngredients = allIngredients.filter(
    (item) =>
      item.minThreshold !== null && item.stockQuantity <= item.minThreshold,
  );

  return {
    todaySales,
    todayOrdersCount,
    totalOrdersCount: allOrders.length,
    totalIngredientsCount: allIngredients.length,
    totalProductsCount: allProducts.length,
    lowStockCount: lowStockIngredients.length,
    lowStockItems: lowStockIngredients.slice(0, 5),
  };
}

"use server";

import { db } from "@/db";
import { expenses } from "@/db/schema";
import { desc } from "drizzle-orm";
import { revalidatePath } from "next/cache";

export async function getExpensesData() {
  const allExpenses = await db.query.expenses.findMany({
    orderBy: [desc(expenses.date)],
  });
  return allExpenses;
}

export async function saveExpense(data: {
  title: string;
  category: string;
  amount: number;
  date: string;
  notes?: string;
}) {
  try {
    const id = "exp_" + Math.random().toString(36).substring(2, 9);
    await db.insert(expenses).values({
      id,
      title: data.title,
      category: data.category,
      amount: data.amount,
      date: data.date,
      notes: data.notes || null,
    });

    revalidatePath("/admin/expenses");
    return { success: true };
  } catch (error) {
    console.error("Failed to save expense:", error);
    return { success: false, error: "Failed to record expense." };
  }
}

"use server";

import { db } from "@/db";
import { ingredients, inventoryLogs } from "@/db/schema";
import { eq } from "drizzle-orm";
import { revalidatePath } from "next/cache";

export async function getInventoryData() {
  const allIngredients = await db.query.ingredients.findMany({
    orderBy: (ingredients, { asc }) => [asc(ingredients.name)],
  });
  return allIngredients;
}

export async function saveIngredient(data: {
  id?: string;
  name: string;
  unit: string;
  costPerUnit: number;
  stockQuantity: number;
  minThreshold: number;
}) {
  const now = new Date().toISOString();

  try {
    await db.transaction(async (tx) => {
      if (data.id) {
        // Update existing ingredient
        const existing = await tx.query.ingredients.findFirst({
          where: eq(ingredients.id, data.id),
        });

        if (existing) {
          const qtyDifference = data.stockQuantity - existing.stockQuantity;

          await tx
            .update(ingredients)
            .set({
              name: data.name,
              unit: data.unit,
              costPerUnit: data.costPerUnit,
              stockQuantity: data.stockQuantity,
              minThreshold: data.minThreshold,
            })
            .where(eq(ingredients.id, data.id));

          // If stock quantity was manually adjusted, log the movement
          if (qtyDifference !== 0) {
            await tx.insert(inventoryLogs).values({
              id: "inv_" + Math.random().toString(36).substring(2, 9),
              ingredientId: data.id,
              type: qtyDifference > 0 ? "adjustment_in" : "adjustment_out",
              quantityChange: qtyDifference,
              reason: "Manual inventory count adjustment",
              createdAt: now,
            });
          }
        }
      } else {
        // Create new ingredient
        const newId = "ing_" + Math.random().toString(36).substring(2, 9);
        await tx.insert(ingredients).values({
          id: newId,
          name: data.name,
          unit: data.unit,
          costPerUnit: data.costPerUnit,
          stockQuantity: data.stockQuantity,
          minThreshold: data.minThreshold,
        });

        if (data.stockQuantity > 0) {
          await tx.insert(inventoryLogs).values({
            id: "inv_" + Math.random().toString(36).substring(2, 9),
            ingredientId: newId,
            type: "initial_stock",
            quantityChange: data.stockQuantity,
            reason: "Initial inventory setup",
            createdAt: now,
          });
        }
      }
    });

    revalidatePath("/admin/inventory");
    revalidatePath("/admin/recipes");
    revalidatePath("/cashier");
    return { success: true };
  } catch (error) {
    console.error("Failed to save ingredient:", error);
    return {
      success: false,
      error: "Failed to save ingredient. Name might already exist.",
    };
  }
}

export async function logWasteOrAdjustment(data: {
  ingredientId: string;
  quantityChange: number; // Negative for waste, positive for find/correction
  reason: string;
}) {
  const now = new Date().toISOString();
  try {
    await db.transaction(async (tx) => {
      const ing = await tx.query.ingredients.findFirst({
        where: eq(ingredients.id, data.ingredientId),
      });

      if (!ing) throw new Error("Ingredient not found");

      const newStock = ing.stockQuantity + data.quantityChange;
      if (newStock < 0) throw new Error("Stock cannot go below zero.");

      await tx
        .update(ingredients)
        .set({ stockQuantity: newStock })
        .where(eq(ingredients.id, data.ingredientId));

      await tx.insert(inventoryLogs).values({
        id: "inv_" + Math.random().toString(36).substring(2, 9),
        ingredientId: data.ingredientId,
        type: data.quantityChange < 0 ? "waste" : "adjustment",
        quantityChange: data.quantityChange,
        reason: data.reason,
        createdAt: now,
      });
    });

    revalidatePath("/admin/inventory");
    return { success: true };
  } catch (error: any) {
    console.error("Failed to log adjustment:", error);
    return {
      success: false,
      error: error.message || "Failed to record adjustment.",
    };
  }
}

"use server";

import { db } from "@/db";
import { products, categories } from "@/db/schema";
import { eq } from "drizzle-orm";
import { revalidatePath } from "next/cache";

export async function getProductsPageData() {
  const allCategories = await db.query.categories.findMany({
    orderBy: (categories, { asc }) => [asc(categories.name)],
  });

  const allProducts = await db.query.products.findMany({
    with: {
      category: true,
      recipeItems: {
        with: {
          ingredient: true,
        },
      },
    },
    orderBy: (products, { asc }) => [asc(products.name)],
  });

  return {
    categories: allCategories,
    products: allProducts,
  };
}

export async function createCategory(name: string) {
  try {
    const id = "cat_" + Math.random().toString(36).substring(2, 9);
    await db.insert(categories).values({ id, name });
    revalidatePath("/admin/products");
    revalidatePath("/admin/recipes");
    revalidatePath("/cashier");
    return { success: true };
  } catch (error) {
    console.error("Failed to create category:", error);
    return { success: false, error: "Category name might already exist." };
  }
}

export async function saveProduct(data: {
  id?: string;
  name: string;
  categoryId: string | null;
  sellingPrice: number;
  isAvailable: boolean;
}) {
  try {
    if (data.id) {
      await db
        .update(products)
        .set({
          name: data.name,
          categoryId: data.categoryId || null,
          sellingPrice: data.sellingPrice,
          isAvailable: data.isAvailable,
        })
        .where(eq(products.id, data.id));
    } else {
      const id = "prod_" + Math.random().toString(36).substring(2, 9);
      await db.insert(products).values({
        id,
        name: data.name,
        categoryId: data.categoryId || null,
        sellingPrice: data.sellingPrice,
        isAvailable: data.isAvailable,
      });
    }

    revalidatePath("/admin/products");
    revalidatePath("/admin/recipes");
    revalidatePath("/cashier");
    return { success: true };
  } catch (error) {
    console.error("Failed to save product:", error);
    return { success: false, error: "Product name might already exist." };
  }
}

export async function toggleProductAvailability(
  id: string,
  currentStatus: boolean,
) {
  try {
    await db
      .update(products)
      .set({ isAvailable: !currentStatus })
      .where(eq(products.id, id));

    revalidatePath("/admin/products");
    revalidatePath("/cashier");
    return { success: true };
  } catch (error) {
    console.error("Failed to toggle availability:", error);
    return { success: false };
  }
}

"use server";

import { db } from "@/db";
import {
  purchases,
  purchaseItems,
  ingredients,
  inventoryLogs,
} from "@/db/schema";
import { eq } from "drizzle-orm";
import { revalidatePath } from "next/cache";

export async function getIngredients() {
  return await db.select().from(ingredients).all();
}

export async function getPurchasesHistory() {
  const result = await db.query.purchases.findMany({
    with: {
      supplier: true,
      items: {
        with: {
          ingredient: true,
        },
      },
    },
    orderBy: (purchases, { desc }) => [desc(purchases.purchasedAt)],
  });
  return result;
}

export async function recordPurchase(data: {
  supplierName?: string;
  totalCost: number;
  notes?: string;
  items: {
    ingredientId: string;
    quantityPurchased: number;
    purchaseUnitName: string;
    unitMultiplier: number;
    totalItemCost: number;
  }[];
}) {
  const purchaseId = "pur_" + Math.random().toString(36).substring(2, 9);
  const now = new Date().toISOString();

  try {
    await db.transaction(async (tx) => {
      // 1. Insert Purchase Record
      await tx.insert(purchases).values({
        id: purchaseId,
        totalCost: data.totalCost,
        purchasedAt: now,
        notes: data.notes || null,
      });

      // 2. Loop through items, insert purchase items, update ingredient stock & cost
      for (const item of data.items) {
        const totalBaseQtyAdded = item.quantityPurchased * item.unitMultiplier;
        const purchaseItemId =
          "pi_" + Math.random().toString(36).substring(2, 9);

        await tx.insert(purchaseItems).values({
          id: purchaseItemId,
          purchaseId: purchaseId,
          ingredientId: item.ingredientId,
          quantityPurchased: item.quantityPurchased,
          purchaseUnitName: item.purchaseUnitName,
          unitMultiplier: item.unitMultiplier,
          totalBaseQuantityAdded: totalBaseQtyAdded,
          totalItemCost: item.totalItemCost,
        });

        // Fetch current ingredient stock
        const currentIng = await tx.query.ingredients.findFirst({
          where: eq(ingredients.id, item.ingredientId),
        });

        if (currentIng) {
          const newStock = currentIng.stockQuantity + totalBaseQtyAdded;
          // Calculate new unit cost (Total cost divided by total base units added in this batch)
          const newCostPerUnit = item.totalItemCost / totalBaseQtyAdded;

          // Update ingredient stock and cost price
          await tx
            .update(ingredients)
            .set({
              stockQuantity: newStock,
              costPerUnit:
                newCostPerUnit > 0 ? newCostPerUnit : currentIng.costPerUnit,
            })
            .where(eq(ingredients.id, item.ingredientId));

          // Log inventory movement
          await tx.insert(inventoryLogs).values({
            id: "inv_" + Math.random().toString(36).substring(2, 9),
            ingredientId: item.ingredientId,
            type: "restock",
            quantityChange: totalBaseQtyAdded,
            referenceId: purchaseId,
            reason: `Restock: ${item.quantityPurchased} ${item.purchaseUnitName}(s)`,
            createdAt: now,
          });
        }
      }
    });

    return { success: true };
  } catch (error) {
    console.error("Failed to record purchase:", error);
    return { success: false, error: "Database transaction failed." };
  }
}

"use server";

import { db } from "@/db";
import { products, ingredients, recipeItems } from "@/db/schema";
import { eq } from "drizzle-orm";
import { revalidatePath } from "next/cache";

export async function getRecipeBuilderData() {
  const allProducts = await db.query.products.findMany({
    with: {
      recipeItems: {
        with: {
          ingredient: true,
        },
      },
    },
    orderBy: (products, { asc }) => [asc(products.name)],
  });

  const allIngredients = await db.query.ingredients.findMany({
    orderBy: (ingredients, { asc }) => [asc(ingredients.name)],
  });

  return {
    products: allProducts,
    ingredients: allIngredients,
  };
}

export async function saveProductRecipe(
  productId: string,
  items: { ingredientId: string; quantity: number }[],
) {
  try {
    await db.transaction(async (tx) => {
      // 1. Remove existing recipe items for this product
      await tx.delete(recipeItems).where(eq(recipeItems.productId, productId));

      // 2. Insert updated recipe items
      if (items.length > 0) {
        const newRecords = items.map((item) => ({
          id: "rec_" + Math.random().toString(36).substring(2, 9),
          productId,
          ingredientId: item.ingredientId,
          quantity: item.quantity,
        }));

        await tx.insert(recipeItems).values(newRecords);
      }
    });

    revalidatePath("/admin/recipes");
    revalidatePath("/cashier");
    return { success: true };
  } catch (error) {
    console.error("Failed to save recipe:", error);
    return { success: false, error: "Failed to save recipe." };
  }
}

interface Ingredient {

  stockQuantity: number;

interface RecipeItem {
  id?: string;
  ingredientId: string;
  quantity: number;

interface ProductWithRecipe {
  id: string;
  name: string;
  sellingPrice: number;
  isAvailable: boolean;
  recipeItems: {
    id: string;
    ingredientId: string;
    quantity: number;
    ingredient: Ingredient;
  }[];
}

export default function RecipesPage() {
  const [products, setProducts] = useState<ProductWithRecipe[]>([]);
  const [ingredients, setIngredients] = useState<Ingredient[]>([]);
  const [searchQuery, setSearchQuery] = useState("");
  const [selectedProductId, setSelectedProductId] = useState<string | null>(
    null,
  );

  // Editor State
  const [draftItems, setDraftItems] = useState<RecipeItem[]>([]);
  const [isSaving, setIsSaving] = useState(false);
  const [successMsg, setSuccessMsg] = useState("");

  useEffect(() => {
    loadData();
  }, []);

  async function loadData() {
    const data = await getRecipeBuilderData();
    setProducts(data.products);
    setIngredients(data.ingredients);

    // Auto-select first product if none selected
    if (data.products.length > 0 && !selectedProductId) {
      selectProduct(data.products[0]);
    } else if (selectedProductId) {
      const updated = data.products.find((p) => p.id === selectedProductId);
      if (updated) selectProduct(updated);
    }
  }

  const selectProduct = (product: ProductWithRecipe) => {
    setSelectedProductId(product.id);
    setSuccessMsg("");
    setDraftItems(
      product.recipeItems.map((ri) => ({
        ingredientId: ri.ingredientId,
        quantity: ri.quantity,
      })),
    );
  };

  const selectedProduct = products.find((p) => p.id === selectedProductId);

  const handleAddIngredientRow = () => {
    setDraftItems([...draftItems, { ingredientId: "", quantity: 1 }]);

    setDraftItems(draftItems.filter((_, i) => i !== index));
  };

    index: number,
    field: keyof RecipeItem,
    value: any,
  ) => {
    const updated = [...draftItems];
    updated[index] = { ...updated[index], [field]: value };
    setDraftItems(updated);
  };

  // Live Cost & Margin Calculations
  const totalFoodCost = draftItems.reduce((sum, item) => {
    const ing = ingredients.find((i) => i.id === item.ingredientId);
    if (!ing) return sum;
    return sum + ing.costPerUnit * (Number(item.quantity) || 0);
  }, 0);

  const sellingPrice = selectedProduct?.sellingPrice || 0;
  const grossProfit = sellingPrice - totalFoodCost;
  const profitMarginPercent =
    sellingPrice > 0 ? (grossProfit / sellingPrice) * 100 : 0;

  const maxServingsPossible =
    draftItems.length > 0 &&

          Math.min(
            ...draftItems.map((item) => {

      setTimeout(() => setSuccessMsg(""), 3500);

"use server";

import { db } from "@/db";
import {
  orders,
  orderItems,
  expenses,
  products,
  recipeItems,
  ingredients,
} from "@/db/schema";
import { desc, eq, sql } from "drizzle-orm";

export async function getFinancialReports() {
  // 1. Fetch all completed orders with their items
  const allOrders = await db.query.orders.findMany({
    where: eq(orders.status, "completed"),
    with: {
      items: {
        with: {
          product: {
            with: {
              recipeItems: {
                with: {
                  ingredient: true,
                },
              },
            },
          },
        },
      },
    },
    orderBy: [desc(orders.createdAt)],
  });

  // 2. Fetch all operational expenses
  const allExpenses = await db.query.expenses.findMany();

  // 3. Compute Metrics
  let totalGrossRevenue = 0;
  let totalEstimatedFoodCost = 0;
  let totalItemSalesCount = 0;

  const productSalesMap: Record<
    string,
    { name: string; qtySold: number; revenue: number }
  > = {};

  // Cast order as any or explicitly type items to satisfy TypeScript iterator check
  for (const order of allOrders as any[]) {
    totalGrossRevenue += order.totalAmount;

    // Ensure order.items is treated as an array
    const items = (order.items || []) as any[];

    for (const item of items) {
      totalItemSalesCount += item.quantity;
      const prodName = item.product?.name || "Unknown Product";

      if (!productSalesMap[item.productId]) {
        productSalesMap[item.productId] = {
          name: prodName,
          qtySold: 0,
          revenue: 0,
        };
      }
      productSalesMap[item.productId].qtySold += item.quantity;
      productSalesMap[item.productId].revenue += item.quantity * item.unitPrice;

      // Calculate food cost for this item sold based on its recipe
      if (item.product?.recipeItems) {
        let singleItemCost = 0;
        for (const ri of item.product.recipeItems) {
          const costPerUnit = ri.ingredient?.costPerUnit || 0;
          singleItemCost += ri.quantity * costPerUnit;
        }
        totalEstimatedFoodCost += singleItemCost * item.quantity;
      }
    }
  }

  const totalOperationalExpenses = allExpenses.reduce(
    (sum, exp) => sum + exp.amount,
    0,
  );
  const grossProfit = totalGrossRevenue - totalEstimatedFoodCost;
  const netProfit = grossProfit - totalOperationalExpenses;
  const netProfitMargin =
    totalGrossRevenue > 0 ? (netProfit / totalGrossRevenue) * 100 : 0;

  const topSellingProducts = Object.values(productSalesMap).sort(
    (a, b) => b.qtySold - a.qtySold,
  );

  return {
    totalGrossRevenue,
    totalEstimatedFoodCost,
    grossProfit,
    totalOperationalExpenses,
    netProfit,
    netProfitMargin,
    totalOrdersCount: allOrders.length,
    totalItemSalesCount,
    topSellingProducts,
    recentOrders: allOrders.slice(0, 10),
    allExpenses: allExpenses.slice(0, 10),
  };
}

  const url = new URL(req.url);
  const type = url.searchParams.get("type");

  if (type === "stream") {
    let currentController: ReadableStreamDefaultController;

    const stream = new ReadableStream({
      start(controller) {
        currentController = controller;
        clients.push(controller);
        controller.enqueue(
          new TextEncoder().encode(
            `data: ${JSON.stringify(terminalCarts)}\n\n`,
          ),
        );
      },
      cancel() {
        // Safely filter out the exact controller instance that disconnected
        clients = clients.filter((c) => c !== currentController);
      },
    });

    return new Response(stream, {
      headers: {
        "Content-Type": "text/event-stream",
        "Cache-Control": "no-cache",
        Connection: "keep-alive",
      },
    });
  }

  return NextResponse.json(terminalCarts);
}

export async function POST(req: Request) {
  try {
    const body = await req.json();
    const { terminalId, ...cartData } = body;
    const targetId = parseInt(terminalId || "1", 10);

    if (terminalCarts[targetId]) {
      terminalCarts[targetId] = {
        ...terminalCarts[targetId],
        ...cartData,
        updatedAt: Date.now(),
      };
    }

    const dataString = `data: ${JSON.stringify(terminalCarts)}\n\n`;
    clients.forEach((client) => {
      try {
        client.enqueue(new TextEncoder().encode(dataString));
      } catch (e) {
        // Disconnected client
      }
    });

    return NextResponse.json({ success: true, terminalCarts });
  } catch (error) {
    return NextResponse.json(
      { success: false, error: "Failed to update cart" },
      { status: 500 },
    );
  }
}

"use server";

import { db } from "@/db";
import {
  products,
  categories,

  orders,
  orderItems,
  inventoryLogs,
} from "@/db/schema";
import { eq } from "drizzle-orm";

export async function getPOSData() {
  const allCategories = await db.query.categories.findMany({
    orderBy: (categories, { asc }) => [asc(categories.name)],
  });

  // Fetch only available products that have recipe items mapped
  const allProducts = await db.query.products.findMany({
    where: eq(products.isAvailable, true),
    with: {
      category: true,
      recipeItems: {
        with: {
          ingredient: true,
        },
      },
    },
    orderBy: (products, { asc }) => [asc(products.name)],
  });

  return {
    categories: allCategories,
    products: allProducts,
  };
}

export async function processPOSOrder(data: {
  paymentMethod: string;
  totalAmount: number;
  items: {
    productId: string;
    quantity: number;
    unitPrice: number;
  }[];
}) {
  const orderId = "ord_" + Math.random().toString(36).substring(2, 9);
  const now = new Date().toISOString();

  try {
    db.transaction((tx) => {
      // 1. Insert Order
      tx.insert(orders)
        .values({
          id: orderId,
          totalAmount: data.totalAmount,
          paymentMethod: data.paymentMethod,
          status: "completed",
          createdAt: now,
        })
        .run();

      // 2. Loop through cart items
      for (const cartItem of data.items) {
        const orderItemId = "oi_" + Math.random().toString(36).substring(2, 9);

        tx.insert(orderItems)
          .values({
            id: orderItemId,
            orderId: orderId,
            productId: cartItem.productId,
            quantity: cartItem.quantity,
            unitPrice: cartItem.unitPrice,
          })
          .run();

        // 3. Fetch product details for logs
        const prodRecord = tx
          .select()
          .from(products)
          .where(eq(products.id, cartItem.productId))
          .get();
        const prodName = prodRecord?.name || "Product";

        // 4. Fetch product recipe items synchronously
        const prodRecipeItems = tx
          .select()
          .from(recipeItems)
          .where(eq(recipeItems.productId, cartItem.productId))
          .all();

        for (const ri of prodRecipeItems) {
          const totalDeduction = ri.quantity * cartItem.quantity;

          // Fetch current ingredient stock synchronously
          const ing = tx
            .select()
            .from(ingredients)
            .where(eq(ingredients.id, ri.ingredientId))
            .get();

          if (ing) {
            const newStock = Math.max(0, ing.stockQuantity - totalDeduction);

            // Update ingredient stock
            tx.update(ingredients)
              .set({ stockQuantity: newStock })
              .where(eq(ingredients.id, ri.ingredientId))
              .run();

            // Log inventory movement
            tx.insert(inventoryLogs)
              .values({
                id: "inv_" + Math.random().toString(36).substring(2, 9),
                ingredientId: ri.ingredientId,
                type: "sale",
                quantityChange: -totalDeduction,
                referenceId: orderId,
                reason: `POS Sale Order #${orderId} (${prodName} x${cartItem.quantity})`,
                createdAt: now,
              })
              .run();
          }
        }
      }
    });

    revalidatePath("/admin/inventory");
    revalidatePath("/admin/reports");
    revalidatePath("/cashier");
    return { success: true, orderId };
  } catch (error) {
    console.error("Failed to process POS order:", error);
    return { success: false, error: "Checkout transaction failed." };
  }
}

        if (tData) {
          setCart(tData.cart || []);
          setTableNumber(tData.tableNumber || "Takeout / Walk-in");

import * as XLSX from "xlsx";

interface ExcelRow {
  Name?: string;
  ProductName?: string;
  Category?: string;

const STORAGE_DIR = path.join(process.cwd(), "data_sessions");

function ensureStorageDir() {
  if (!fs.existsSync(STORAGE_DIR)) {
    fs.mkdirSync(STORAGE_DIR, { recursive: true });
  }

export async function getIndependentMenuData() {
  try {
    const filePath = path.join(process.cwd(), "menu.xlsx");
    if (!fs.existsSync(filePath)) {
      return { categories: [], products: [] };
    }

    const fileBuffer = fs.readFileSync(filePath);
    const workbook = XLSX.read(fileBuffer, { type: "buffer" });
    const sheetName = workbook.SheetNames[0];
    const sheetData = XLSX.utils.sheet_to_json<ExcelRow>(
      workbook.Sheets[sheetName],
    );

    const categoriesMap = new Map<string, { id: string; name: string }>();
    const products = sheetData.map((row, index) => {
      const categoryName = row.Category || "General";
      const catId = categoryName.toLowerCase().replace(/\s+/g, "_");

      if (!categoriesMap.has(catId)) {
        categoriesMap.set(catId, { id: catId, name: categoryName });
      }

      return {
        id: `xlsx_item_${index + 1}`,
        name: row.Name || row.ProductName || "Unnamed Item",
        sellingPrice: Number(row.Price) || 0,
        categoryId: catId,
        category: { id: catId, name: categoryName },
      };
    });

    return {
      categories: Array.from(categoriesMap.values()),
      products,
    };
  } catch (error) {
    console.error("Error reading menu.xlsx:", error);
    return { categories: [], products: [] };
  }
}

// Save order to current active session
export async function saveIndependentOrder(orderData: {
  paymentMethod: string;
  tableNumber?: string;
  totalAmount: number;
  items: any[];
}) {
  try {
    ensureStorageDir();
    const activeSessionPath = path.join(STORAGE_DIR, "active_session.json");

    let sessionData = {
      sessionId:
        "session_" +
        new Date().toISOString().split("T")[0] +
        "_" +
        Math.random().toString(36).substring(2, 6),
      date: new Date().toISOString().split("T")[0],
      orders: [] as any[],
    };

    if (fs.existsSync(activeSessionPath)) {
      sessionData = JSON.parse(fs.readFileSync(activeSessionPath, "utf-8"));
    }

    const queueNumber = sessionData.orders.length + 1;

    const orderRecord = {
      orderId: "xlsx_ord_" + Math.random().toString(36).substring(2, 9),
      queueNumber,
      tableNumber: orderData.tableNumber || "Takeout / Walk-in",
      timestamp: new Date().toISOString(),
      ...orderData,
    };

    sessionData.orders.push(orderRecord);
    fs.writeFileSync(activeSessionPath, JSON.stringify(sessionData, null, 2));

    return { success: true, orderId: orderRecord.orderId, queueNumber };
  } catch (error) {
    console.error("Failed to save independent order:", error);
    return { success: false, error: "Failed to save order." };
  }
}

export async function getActiveSessionData() {
  try {
    ensureStorageDir();
    const activeSessionPath = path.join(STORAGE_DIR, "active_session.json");
    if (!fs.existsSync(activeSessionPath)) {
      return {

        date: new Date().toISOString().split("T")[0],
        orders: [],
      };
    }
    return JSON.parse(fs.readFileSync(activeSessionPath, "utf-8"));
  } catch (error) {

// Clear active session entirely
export async function restartSession() {
  try {
    ensureStorageDir();
    const activeSessionPath = path.join(STORAGE_DIR, "active_session.json");
    if (fs.existsSync(activeSessionPath)) {
      fs.unlinkSync(activeSessionPath);
    }
    return { success: true };
  } catch (error) {
    console.error("Failed to restart session:", error);
    return { success: false, error: "Failed to restart session." };
  }
}

// Delete a specific order from the active session by orderId
export async function deleteActiveSessionOrder(orderId: string) {
  try {
    ensureStorageDir();
    const activeSessionPath = path.join(STORAGE_DIR, "active_session.json");
    if (fs.existsSync(activeSessionPath)) {
      let sessionData = JSON.parse(fs.readFileSync(activeSessionPath, "utf-8"));
      if (sessionData.orders) {
        sessionData.orders = sessionData.orders.filter(
          (o: any) => o.orderId !== orderId,
        );
        fs.writeFileSync(
          activeSessionPath,
          JSON.stringify(sessionData, null, 2),
        );
      }
    }
    return { success: true };
  } catch (error) {
    console.error("Failed to delete active session order:", error);
    return { success: false, error: "Failed to delete order." };
  }
}

// Export current active session or specified date orders
export async function exportSessionByDate(dateStr: string) {
  try {
    ensureStorageDir();
    const activeSession = await getActiveSessionData();

    const matchingSessions = [];
    if (activeSession && activeSession.orders?.length > 0) {
      matchingSessions.push(activeSession);
    }

    if (matchingSessions.length === 0) {
      return {
        success: false,
        error: "No orders found in the active session.",
      };
    }

    const combinedOrders: any[] = [];
    matchingSessions.forEach((session: any) => {
      if (session.orders) {
        combinedOrders.push(...session.orders);
      }
    });

    const excelRows: any[] = [];
    combinedOrders.forEach((ord) => {
      ord.items.forEach((item: any) => {
        excelRows.push({
          Date: ord.timestamp ? ord.timestamp.split("T")[0] : dateStr,
          "Session ID": ord.orderId ? ord.orderId.substring(0, 15) : "session",
          "Queue #": ord.queueNumber || 1,
          "Table / Location": ord.tableNumber || "Takeout / Walk-in",
          "Order ID": ord.orderId,
          "Product Name": item.name,
          Quantity: item.quantity,
          "Unit Price (â‚±)": item.unitPrice,
          "Line Total (â‚±)": item.quantity * item.unitPrice,
          "Payment Method": ord.paymentMethod,
          "Order Total (â‚±)": ord.totalAmount,
          Timestamp: ord.timestamp,
        });
      });
    });

    const worksheet = XLSX.utils.json_to_sheet(excelRows);
    const workbook = XLSX.utils.book_new();
    XLSX.utils.book_append_sheet(workbook, worksheet, `Sales_${dateStr}`);

    const exportFileName = `Active_Session_Export_${dateStr}.xlsx`;

    // Write workbook directly to a Node.js Buffer instead of writing to disk
    const excelBuffer = XLSX.write(workbook, {
      bookType: "xlsx",
      type: "buffer",
    });
    const base64Data = Buffer.from(excelBuffer).toString("base64");

    return {
      success: true,
      fileName: exportFileName,
      fileData: base64Data,
    };
  } catch (error) {
    console.error("Export failed:", error);
    return { success: false, error: "Failed to generate Excel export." };
  }
}

  orderItems,
  inventoryLogs,
} from "@/db/schema";
import { eq } from "drizzle-orm";
import { revalidatePath } from "next/cache";

export async function getPOSData() {
  const allCategories = await db.query.categories.findMany({
    orderBy: (categories, { asc }) => [asc(categories.name)],
  });

  // Fetch only available products that have recipe items mapped
  const allProducts = await db.query.products.findMany({
    where: eq(products.isAvailable, true),
    with: {
      category: true,
      recipeItems: {
        with: {
          ingredient: true,
        },
      },
    },
    orderBy: (products, { asc }) => [asc(products.name)],
  });

  return {
    categories: allCategories,
    products: allProducts,
  };
}

export async function processPOSOrder(data: {
  paymentMethod: string;
  totalAmount: number;
  items: {
    productId: string;
    quantity: number;
    unitPrice: number;
  }[];
}) {
  const orderId = "ord_" + Math.random().toString(36).substring(2, 9);
  const now = new Date().toISOString();

  try {
    await db.transaction(async (tx) => {
      // 1. Insert Order
      await tx.insert(orders).values({
        id: orderId,
        totalAmount: data.totalAmount,
        paymentMethod: data.paymentMethod,
        status: "completed",
        createdAt: now,
      });

      // 2. Loop through cart items
      for (const cartItem of data.items) {
        const orderItemId = "oi_" + Math.random().toString(36).substring(2, 9);

        await tx.insert(orderItems).values({
          id: orderItemId,
          orderId: orderId,
          productId: cartItem.productId,
          quantity: cartItem.quantity,
          unitPrice: cartItem.unitPrice,
        });

        // 3. Fetch product recipe to deduct ingredients
        const prod = await tx.query.products.findFirst({
          where: eq(products.id, cartItem.productId),
          with: {
            recipeItems: true,
          },
        });

        if (prod && prod.recipeItems) {
          for (const ri of prod.recipeItems) {
            const totalDeduction = ri.quantity * cartItem.quantity;

            // Fetch current ingredient stock
            const ing = await tx.query.ingredients.findFirst({
              where: eq(ingredients.id, ri.ingredientId),
            });

            if (ing) {
              const newStock = Math.max(0, ing.stockQuantity - totalDeduction);

              // Update ingredient stock
              await tx
                .update(ingredients)
                .set({ stockQuantity: newStock })
                .where(eq(ingredients.id, ri.ingredientId));

              // Log inventory movement
              await tx.insert(inventoryLogs).values({
                id: "inv_" + Math.random().toString(36).substring(2, 9),
                ingredientId: ri.ingredientId,
                type: "sale",
                quantityChange: -totalDeduction,
                referenceId: orderId,
                reason: `POS Sale Order #${orderId} (${prod.name} x${cartItem.quantity})`,
                createdAt: now,
              });
            }
          }
        }
      }
    });

    revalidatePath("/admin/inventory");
    revalidatePath("/admin/reports");
    revalidatePath("/cashier");
    return { success: true, orderId };
  } catch (error) {
    console.error("Failed to process POS order:", error);
    return { success: false, error: "Checkout transaction failed." };
  }
}

import fs from "fs";
import path from "path";

// Update this filename if your src/db/index.ts points to a different .db file name
const DB_FILE_NAME = "sqlite.db";

async function backupDatabase() {
  const rootDir = process.cwd();
  const dbPath = path.join(rootDir, DB_FILE_NAME);
  const backupDir = path.join(rootDir, "backups");

  if (!fs.existsSync(dbPath)) {
    console.error(`❌ Database file not found at: ${dbPath}`);
    console.info("Check src/db/index.ts to verify your SQLite filename.");
    process.exit(1);
  }

  // Create backups/ directory if it doesn't exist
  if (!fs.existsSync(backupDir)) {
    fs.mkdirSync(backupDir, { recursive: true });
  }

  // Generate timestamped filename: e.g., sqlite-backup-2026-11-13T06-28-00.db
  const timestamp = new Date().toISOString().replace(/[:.]/g, "-").slice(0, 19);
  const backupFileName = `sqlite-backup-${timestamp}.db`;
  const backupPath = path.join(backupDir, backupFileName);

  // Copy main SQLite database file
  fs.copyFileSync(dbPath, backupPath);

  // Also copy WAL / SHM files if SQLite WAL mode is enabled
  for (const ext of ["-wal", "-shm"]) {
    const journalPath = `${dbPath}${ext}`;
    if (fs.existsSync(journalPath)) {
      fs.copyFileSync(journalPath, `${backupPath}${ext}`);
    }
  }

  console.log(
    `✅ Database backed up successfully to: backups/${backupFileName}`,
  );
}

backupDatabase();

import { sqliteTable, text, real, integer } from "drizzle-orm/sqlite-core";
import { relations } from "drizzle-orm";

// ==========================================
// 1. INVENTORY & RAW MATERIALS
// ==========================================
export const ingredients = sqliteTable("ingredients", {
  id: text("id").primaryKey(),
  name: text("name").notNull().unique(),
  unit: text("unit").notNull(), // Base unit used in recipes: 'kg', 'g', 'ml', 'pcs'
  costPerUnit: real("cost_per_unit").notNull(), // Cost per 1 base unit
  stockQuantity: real("stock_quantity").notNull().default(0), // Current physical stock in base units
  minThreshold: real("min_threshold").default(0), // Low-stock warning level
});

// ==========================================
// 2. MENU CATEGORIES, PRODUCTS & RECIPES
// ==========================================
export const categories = sqliteTable("categories", {
  id: text("id").primaryKey(),
  name: text("name").notNull().unique(), // e.g., "Burgers", "Rice Meals", "Drinks"
});

export const products = sqliteTable("products", {
  id: text("id").primaryKey(),
  categoryId: text("category_id").references(() => categories.id, {
    onDelete: "set null",
  }),
  name: text("name").notNull().unique(),
  sellingPrice: real("selling_price").notNull(),
  isAvailable: integer("is_available", { mode: "boolean" })
    .notNull()
    .default(true),
});

// Bill of Materials (BOM): Links Products to Ingredients
export const recipeItems = sqliteTable("recipe_items", {
  id: text("id").primaryKey(),
  productId: text("product_id")
    .notNull()
    .references(() => products.id, { onDelete: "cascade" }),
  ingredientId: text("ingredient_id")
    .notNull()
    .references(() => ingredients.id, { onDelete: "restrict" }),
  quantity: real("quantity").notNull(), // Amount of base unit used per 1 product sold
});

// ==========================================
// 3. STOCK-IN (SUPPLIERS & PURCHASES)
// ==========================================
export const suppliers = sqliteTable("suppliers", {
  id: text("id").primaryKey(),
  name: text("name").notNull(),
  contactNumber: text("contact_number"),
});

export const purchases = sqliteTable("purchases", {
  id: text("id").primaryKey(),
  supplierId: text("supplier_id").references(() => suppliers.id, {
    onDelete: "set null",
  }),
  totalCost: real("total_cost").notNull(),
  purchasedAt: text("purchased_at").notNull(), // ISO timestamp (e.g., new Date().toISOString())
  notes: text("notes"),
});

export const purchaseItems = sqliteTable("purchase_items", {
  id: text("id").primaryKey(),
  purchaseId: text("purchase_id")
    .notNull()
    .references(() => purchases.id, { onDelete: "cascade" }),
  ingredientId: text("ingredient_id")
    .notNull()
    .references(() => ingredients.id, { onDelete: "restrict" }),
  quantityPurchased: real("quantity_purchased").notNull(), // e.g., 2 (sacks), 5 (packs)
  purchaseUnitName: text("purchase_unit_name").notNull(), // e.g., 'sack', 'pack', 'pcs'
  unitMultiplier: real("unit_multiplier").notNull(), // e.g., 25 (for 25kg sack), 6 (for pack of 6)
  totalBaseQuantityAdded: real("total_base_quantity_added").notNull(), // quantityPurchased * unitMultiplier
  totalItemCost: real("total_item_cost").notNull(), // Total cost for this line item
});

// ==========================================
// 4. STOCK-OUT (POS SALES & ORDERS)
// ==========================================
export const orders = sqliteTable("orders", {
  id: text("id").primaryKey(),
  totalAmount: real("total_amount").notNull(),
  paymentMethod: text("payment_method").notNull(), // e.g., 'cash', 'gcash', 'maya', 'card'
  status: text("status").notNull().default("completed"), // 'completed', 'voided'
  createdAt: text("created_at").notNull(), // ISO timestamp
});

export const orderItems = sqliteTable("order_items", {
  id: text("id").primaryKey(),
  orderId: text("order_id")
    .notNull()
    .references(() => orders.id, { onDelete: "cascade" }),
  productId: text("product_id")
    .notNull()
    .references(() => products.id, { onDelete: "restrict" }),
  quantity: integer("quantity").notNull(),
  unitPrice: real("unit_price").notNull(), // Snapshot of sellingPrice at the time of sale
});

// ==========================================
// 5. INVENTORY AUDIT LOGS & OPERATIONAL EXPENSES
// ==========================================
export const inventoryLogs = sqliteTable("inventory_logs", {
  id: text("id").primaryKey(),
  ingredientId: text("ingredient_id")

  type: text("type").notNull(), // 'restock', 'sale', 'waste', 'adjustment'
  quantityChange: real("quantity_change").notNull(), // (+) for restock, (-) for sale/waste
  referenceId: text("reference_id"), // Optional: links to orderId or purchaseId

export const expenses = sqliteTable("expenses", {
